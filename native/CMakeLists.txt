option(ASTRA_BUILD_PLUGIN "Build actual SKSE plugin" ON)
# Variant for Skyrim AE 1.7.104 (SKSE 2.3.1, Address Library v13): built against the maintained alandtse fork of
# CommonLibSSE-NG, licensed GPL-3.0-or-later with the modding/linking exceptions (EXCEPTIONS.md). Its dependencies come
# from vcpkg (CMAKE_TOOLCHAIN_FILE). The default build (1.6.1170, CharmedBaryon fork, MIT) is unchanged.
option(ASTRA_AE17 "Build the AstraSight variant for Skyrim 1.7.104" OFF)
# Variant for Skyrim SE 1.5.97 (SKSE 2.0.20, Address Library v1): same pinned MIT CommonLibSSE-NG, SE layouts.
option(ASTRA_SE1597 "Build the AstraSight variant for Skyrim 1.5.97" OFF)
if(ASTRA_BUILD_PLUGIN AND ASTRA_AE17)
  set(ASTRA_VENDOR "${CMAKE_CURRENT_SOURCE_DIR}/../build/native-ae17" CACHE PATH "1.7.104 dependency sources")
  set(BUILD_TESTS OFF CACHE BOOL "" FORCE)
  set(ENABLE_SKYRIM_SE OFF CACHE BOOL "" FORCE)
  set(ENABLE_SKYRIM_VR OFF CACHE BOOL "" FORCE)
  set(ENABLE_SKYRIM_AE ON CACHE BOOL "" FORCE)
  add_subdirectory("${ASTRA_VENDOR}/CommonLibSSE-NG" "${CMAKE_CURRENT_BINARY_DIR}/CommonLibSSE")
  add_library(AstraSight SHARED src/Plugin.cpp src/FastShotRuntime.cpp src/GuardRuntime.cpp)
  target_link_libraries(AstraSight PRIVATE AstraCore CommonLibSSE::CommonLibSSE)
  target_compile_definitions(AstraSight PRIVATE ASTRA_RUNTIME_AE17=1 NOMINMAX)
  target_precompile_headers(AstraSight PRIVATE "${ASTRA_VENDOR}/CommonLibSSE-NG/include/SKSE/Impl/PCH.h")
elseif(ASTRA_BUILD_PLUGIN)
  set(ASTRA_VENDOR "${CMAKE_CURRENT_SOURCE_DIR}/../build/native" CACHE PATH "Pinned native dependency sources (read-only when external)")
  set(SPDLOG_BUILD_TESTS OFF CACHE BOOL "" FORCE)
  set(SPDLOG_INSTALL ON CACHE BOOL "" FORCE)
  add_subdirectory("${ASTRA_VENDOR}/spdlog" "${CMAKE_CURRENT_BINARY_DIR}/spdlog")
  file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/package-shims")
  file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/package-shims/spdlogConfig.cmake" "if(NOT TARGET spdlog::spdlog)\n message(FATAL_ERROR \"Pinned spdlog target missing\")\nendif()\nset(spdlog_FOUND TRUE)\n")
  set(spdlog_DIR "${CMAKE_CURRENT_BINARY_DIR}/package-shims")
  set(RAPIDCSV_INCLUDE_DIRS "${ASTRA_VENDOR}/rapidcsv/src" CACHE PATH "" FORCE)
  set(BUILD_TESTS OFF CACHE BOOL "" FORCE)
  if(ASTRA_SE1597)
    set(ENABLE_SKYRIM_SE ON CACHE BOOL "" FORCE)
    set(ENABLE_SKYRIM_AE OFF CACHE BOOL "" FORCE)
  else()
    set(ENABLE_SKYRIM_SE OFF CACHE BOOL "" FORCE)
    set(ENABLE_SKYRIM_AE ON CACHE BOOL "" FORCE)
  endif()
  set(ENABLE_SKYRIM_VR OFF CACHE BOOL "" FORCE)
  add_subdirectory("${ASTRA_VENDOR}/CommonLibSSE-NG" "${CMAKE_CURRENT_BINARY_DIR}/CommonLibSSE")
  add_library(AstraSight SHARED src/Plugin.cpp src/FastShotRuntime.cpp src/GuardRuntime.cpp)
  target_link_libraries(AstraSight PRIVATE AstraCore CommonLibSSE::CommonLibSSE)
  target_precompile_headers(AstraSight PRIVATE "${ASTRA_VENDOR}/CommonLibSSE-NG/include/SKSE/Impl/PCH.h")
  if(ASTRA_SE1597)
    target_compile_definitions(AstraSight PRIVATE ASTRA_RUNTIME_SE1597=1)
  endif()
endif()
