cmake_minimum_required(VERSION 3.26)
project(AstraSight VERSION 3.3.0 LANGUAGES CXX)
# Installed compiler emits UTF-8 Spanish include diagnostics. CMake's probe
# misdecodes that prefix; correct it so Ninja tracks headers and avoids stale ABI.
if(MSVC)
  set(CMAKE_CL_SHOWINCLUDES_PREFIX "Nota: inclusión del archivo: ")
endif()
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(AstraCore STATIC src/Core.cpp)
target_include_directories(AstraCore PUBLIC include)
enable_testing()
add_executable(AstraCoreTests tests/core_tests.cpp)
target_link_libraries(AstraCoreTests PRIVATE AstraCore)
add_test(NAME AstraCoreTests COMMAND AstraCoreTests)
add_executable(AstraV07Tests tests/v07_tests.cpp)
target_link_libraries(AstraV07Tests PRIVATE AstraCore)
add_test(NAME AstraV07Tests COMMAND AstraV07Tests)
add_executable(AstraV08Tests tests/v08_tests.cpp)
target_link_libraries(AstraV08Tests PRIVATE AstraCore)
add_test(NAME AstraV08Tests COMMAND AstraV08Tests)
add_executable(AstraV09Tests tests/v09_tests.cpp)
target_link_libraries(AstraV09Tests PRIVATE AstraCore)
add_test(NAME AstraV09Tests COMMAND AstraV09Tests)
add_executable(AstraV10Tests tests/v10_tests.cpp)
target_link_libraries(AstraV10Tests PRIVATE AstraCore)
add_test(NAME AstraV10Tests COMMAND AstraV10Tests)
option(ASTRA_BUILD_PLUGIN "Build actual SKSE plugin" ON)
if(ASTRA_BUILD_PLUGIN)
  set(ASTRA_VENDOR "${CMAKE_CURRENT_SOURCE_DIR}/../build/native")
  set(SPDLOG_BUILD_TESTS OFF CACHE BOOL "" FORCE)
  set(SPDLOG_INSTALL ON CACHE BOOL "" FORCE)
  add_subdirectory("${ASTRA_VENDOR}/spdlog" "${CMAKE_CURRENT_BINARY_DIR}/spdlog")
  file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/package-shims")
  file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/package-shims/spdlogConfig.cmake" "if(NOT TARGET spdlog::spdlog)\n message(FATAL_ERROR \"Pinned spdlog target missing\")\nendif()\nset(spdlog_FOUND TRUE)\n")
  set(spdlog_DIR "${CMAKE_CURRENT_BINARY_DIR}/package-shims")
  set(RAPIDCSV_INCLUDE_DIRS "${ASTRA_VENDOR}/rapidcsv/src" CACHE PATH "" FORCE)
  set(BUILD_TESTS OFF CACHE BOOL "" FORCE)
  set(ENABLE_SKYRIM_SE OFF CACHE BOOL "" FORCE)
  set(ENABLE_SKYRIM_VR OFF CACHE BOOL "" FORCE)
  set(ENABLE_SKYRIM_AE ON CACHE BOOL "" FORCE)
  add_subdirectory("${ASTRA_VENDOR}/CommonLibSSE-NG" "${CMAKE_CURRENT_BINARY_DIR}/CommonLibSSE")
  add_library(AstraSight SHARED src/Plugin.cpp)
  target_link_libraries(AstraSight PRIVATE AstraCore CommonLibSSE::CommonLibSSE)
  target_precompile_headers(AstraSight PRIVATE "${ASTRA_VENDOR}/CommonLibSSE-NG/include/SKSE/Impl/PCH.h")
endif()
